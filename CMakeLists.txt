cmake_minimum_required(VERSION 3.16)
project(ClubManager VERSION 3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(club_core src/hash_map.cpp src/avl.cpp src/friend_graph.cpp
  src/inverted_index.cpp src/system.cpp src/features.cpp src/data_io.cpp src/persistence.cpp)
target_include_directories(club_core PUBLIC include)
find_package(Threads REQUIRED)
add_library(club_auth src/auth.cpp)
target_link_libraries(club_auth PUBLIC club_core)
if(WIN32)
  target_link_libraries(club_auth PUBLIC bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(club_auth PUBLIC OpenSSL::Crypto)
endif()
add_executable(club src/main.cpp src/web_server.cpp)
target_link_libraries(club PRIVATE club_core club_auth Threads::Threads)
if(WIN32)
  target_link_libraries(club PRIVATE ws2_32 shell32)
endif()
enable_testing()
add_executable(club_tests tests/test_main.cpp)
target_link_libraries(club_tests PRIVATE club_core)
add_test(NAME unit COMMAND club_tests)
set_tests_properties(unit PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  add_test(NAME http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_http.py" "$<TARGET_FILE:club>")
  add_test(NAME auth COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_auth.py" "$<TARGET_FILE:club>")
endif()
